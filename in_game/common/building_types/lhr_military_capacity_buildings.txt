# Frontier Fort - a capped Hungarian border fortification that also expands military capacity
lhr_frontier_fort = {
	is_foreign = no
	pop_type = soldiers
	max_levels = 1

	category = defense_category
	forbidden_for_estates = yes
	important_for_AI = yes

	employment_size = castle_employment

	rural_settlement = yes
	town = yes
	city = yes

	build_time = medium_fort_building

	expensive = yes

	allow = {
		owner = {
			has_advance = hun_castle_warriors
		}
		OR = {
			region = region:balkan_region
			area = area:south_alfold_area
			area = area:wallachia_area
			area = area:transylvania_area
		}
		NOT = { has_building = building_type:stockade }
		NOT = { has_building = building_type:castle }
		NOT = { has_building = building_type:bastion }
		NOT = { has_building = building_type:star_fort }
		NOT = { has_building = building_type:fortress }

		custom_tooltip = {
			text = lhr_frontier_fort_tt
			scope:actor = {	total_building_levels_including_construction:lhr_frontier_fort < modifier:lhr_frontier_fort_limit }
		}
		owner = scope:actor
	}

	unique_production_methods = {
		lhr_frontier_fort_maintenance = {
			stone = 0.12
			tools = 0.05
			weaponry = 0.12
			tar = 0.03

			category = building_maintenance
		}
	}
	construction_demand = castle_construction

	modifier = {
		local_garrison_size = 0.7
		local_unrest = -0.05
		lhr_local_military_capacity = 500
		local_manpower = 0.01
	}

	raw_modifier = {
		fort_level = 2
		local_nobles_desired_pop = 0.02
		propagating_zone_of_control = yes
	}
}

# Swedish Allotment - one regional military-administrative allotment node per region
lhr_swedish_allotment = {
	is_foreign = no
	pop_type = soldiers
	max_levels = 1

	category = military_category
	forbidden_for_estates = yes

	employment_size = 0.1
	important_for_AI = yes
	AI_ignore_available_worker_flag = yes	
	rural_settlement = yes
	town = yes
	city = no

	build_time = soldier_building

	allow = {
		save_temporary_scope_as = lhr_allotment_target_location
		owner = {
			has_advance = indelningsverket
			NOT = {
				any_owned_location = {
					region = scope:lhr_allotment_target_location.region
					has_building = building_type:lhr_swedish_allotment
				}
			}
		}

		custom_tooltip = {
			text = lhr_swedish_allotment_region_tt
			scope:actor = {
				NOT = {
					any_owned_location = {
						region = scope:lhr_allotment_target_location.region
						has_building = building_type:lhr_swedish_allotment
					}
				}
			}
		}
		owner = scope:actor
	}

	remove_if = {
		owner = {
			NOT = { has_or_had_tag = SWE }
		}
	}

	modifier = {
		lhr_local_military_capacity = 250
		local_manpower = 0.005
	}

	unique_production_methods = {
		lhr_swedish_allotment_maintenance = {
			grain = 0.35
			leather = 0.15
			tools = 0.15

			category = building_maintenance
		}
	}
}
