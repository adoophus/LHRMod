	on_actions = {

yearly_country_pulse = {
	on_actions = {
		mod_apply_no_father_on_yearly_pulse
	}
}

mod_apply_no_father_on_yearly_pulse = {
	effect = {
		if = {
			limit = {
				exists = capital
				NOT = { has_country_modifier = lhr_chi_permanent_nerf }
				OR = {
					capital = { region = region:south_china_region }
					capital = { region = region:east_china_region }
					capital = { region = region:north_china_region }
				}
			}
			add_country_modifier = { modifier = lhr_chi_permanent_nerf months = -1 mode = add }
		}
		else = {
			remove_country_modifier = lhr_chi_permanent_nerf
		}
	}
}

mod_run_performance_cleanup = {
	effect = {
		every_country = {
			limit = {
				country_type = pop
			}
			change_country_type = building
			every_owned_building = {
				change_building_level = -10
			}
		}

		sub_continent:pacific_islands = { mod_remove_and_disable_sub_continent = yes }
		sub_continent:east_asia = { mod_remove_and_disable_sub_continent = yes }
		sub_continent:south_east_asia = { mod_remove_and_disable_sub_continent = yes }
		
	}
}

mod_nerf_bohemia = {

		c:BOH = {
			cancel_subject = c:MVA
			cancel_subject = c:OPL

			cancel_subject = c:GLO

			create_relation = {
				first = c:BOH
				second = c:WRC
				type = relation_type:guarantee
			}

			create_relation = {
				first = c:BOH
				second = c:GLO
				type = relation_type:guarantee

		}

		c:NYS = {
			make_subject_of = {
				target = c:BOH
				type = subject_type:vassal
			}
		}

		c:ZAG = {
			make_subject_of = {
				target = c:GLO

			}
		}
	}
}
